Recordings Finder

Search hundreds of recorded conference talks, webinars, workshops, network huddles and project showcases from across the health and care analytics community, all in one place.

Where a long recording covers several talks and its YouTube description gives the timings, each talk is listed separately and its link starts the video at that talk. Each recording also links to its Atlas entry, where you can find out more about the community that published it.

recordings = (await FileAttachment("recordings.json").json())
  .map((d, i) => ({...d, order: i}))
  .sort((a, b) => (b.year ?? 0) - (a.year ?? 0) || b.date.localeCompare(a.date) || a.order - b.order)

entryLinks = new Map(recordings.map(d => [d.collection, d.collection_url]))

function shorten(text, length = 160) {
  return text.length > length ? text.slice(0, length).replace(/\s+\S*$/, "") + "…" : text;
}

function options(values, all) {
  return [all, ...Array.from(new Set(values)).sort((a, b) => a < b ? -1 : a > b ? 1 : 0)];
}
viewof query = Inputs.search(recordings, {
  label: "Search",
  placeholder: "Search titles, speakers and topics",
  width: 420
})
viewof source = Inputs.select(options(recordings.map(d => d.source), "All sources"), {label: "Source", width: 240})
viewof type = Inputs.select(options(recordings.map(d => d.type), "All types"), {label: "Type", width: 240})
viewof year = Inputs.select(
  ["All years", ...options(recordings.map(d => d.year).filter(d => d != null), "").slice(1).reverse()],
  {label: "Year", width: 240}
)
filtered = query.filter(d =>
  (source === "All sources" || d.source === source) &&
  (type === "All types" || d.type === type) &&
  (year === "All years" || d.year === year)
)

md`Showing **${filtered.length}** of ${recordings.length} recordings.`
html`<div class="recordings-table">${Inputs.table(filtered, {
  columns: ["title", "year", "type", "event", "source", "details", "url", "collection"],
  header: {
    title: "Title",
    year: "Year",
    type: "Type",
    event: "Event",
    source: "Source",
    details: "Details",
    url: "Watch",
    collection: "More"
  },
  format: {
    year: d => d == null ? "" : String(d),
    details: d => shorten(d),
    url: d => html`<a href="${d}" target="_blank" rel="noopener">Watch <i class="bi bi-box-arrow-up-right"></i></a>`,
    collection: d => html`<a href="${entryLinks.get(d)}" title="${d}">Atlas entry</a>`
  },
  width: {title: 280, year: 60, type: 120, event: 110, source: 120, details: 330, url: 80, collection: 100},
  layout: "fixed",
  rows: 25,
  height: 750,
  select: false
})}</div>`
